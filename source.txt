local ProtectionConfig = {
    -- 🔴 CRITICAL: This MUST exactly match the 'Secret' value in your Key System's Config!
    -- If your Key System has: Secret = "Test"
    -- Then this must also be: SecretKey = "Test"
    SecretKey = "Settjopakalala",
    
    -- The name of your Hub (shown in the kick message if they try to bypass)
    HubName = "Rowe Hub"
}

-- Anti-Bypass Logic: Checks if the Key System successfully set the global variable
if not _G[ProtectionConfig.SecretKey] then
    local player = game:GetService("Players").LocalPlayer
    if player then
        player:Kick("\n🛡️ Unauthorized Execution 🛡️\n\nPlease use the official Key System to run " .. ProtectionConfig.HubName)
    end
    return -- Stops the rest of the script from loading!
end

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local VirtualUser = game:GetService("VirtualUser") -- Добавлено для Anti-AFK

local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

local WaveRemotes = ReplicatedStorage:WaitForChild("WaveRemotes")
local SkipVote = WaveRemotes:WaitForChild("SkipVote")
local GameStateUpdate = WaveRemotes:WaitForChild("GameStateUpdate")

local ZombieRemotes = ReplicatedStorage:WaitForChild("ZombieRemotes")
local ZombieDamage = ZombieRemotes:WaitForChild("ZombieDamage")
local ZombieSpawn = ZombieRemotes:WaitForChild("ZombieSpawn")

local zombiesFolder = Workspace:WaitForChild("Zombies_Local")

local instakillEnabled = false

-- ⚙️ НАСТРОЙКА СЕТИ (Плавный поток без спама пачками)
local DELAY_BETWEEN_KILLS = 0.04 -- Задержка в секундах между убийствами (если лагает — поставь 0.06)

-- Поиск спавна
local function GetSafeSpawnPos()
    local spawnPart = Workspace:FindFirstChildWhichIsA("SpawnLocation", true)
    if spawnPart then
        return spawnPart.Position + Vector3.new(0, 25, 0)
    end
    return Vector3.new(-257.93, 505.37, -416.37)
end

-- 1. Зависание над Спавн-Локацией
if _G.FreezeConnection then _G.FreezeConnection:Disconnect() end
_G.FreezeConnection = RunService.Heartbeat:Connect(function()
    if not instakillEnabled then return end
    
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if root then
        root.CFrame = CFrame.new(GetSafeSpawnPos())
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end
end)

-- 2. Авто-пропуск волны
if _G.SkipWaveConnection then _G.SkipWaveConnection:Disconnect() end
_G.SkipWaveConnection = GameStateUpdate.OnClientEvent:Connect(function(data)
    if data == "WaveStarted" and instakillEnabled then
        task.wait(0.5)
        SkipVote:FireServer(true)
    end
end)

-- 3. УМНЫЙ МАСС-КИЛЛ (Бьет и тех, кто уже на карте)
if _G.TurboKillTask then _G.TurboKillTask = false end
_G.TurboKillTask = true

task.spawn(function()
    while _G.TurboKillTask do
        if instakillEnabled then
            local children = zombiesFolder:GetChildren()
            
            for i = 1, #children do
                if not instakillEnabled then break end
                
                local obj = children[i]
                if obj:IsA("Model") then
                    -- Ищем ID в имени модели (поддерживает "Zombie_123" и просто "123")
                    local idStr = obj.Name:match("Zombie_(%d+)") or obj.Name:match("(%d+)")
                    local zombieId = idStr and tonumber(idStr)
                    
                    -- Если в имени нет, проверяем атрибуты модели
                    if not zombieId then
                        zombieId = obj:GetAttribute("ZombieId") or obj:GetAttribute("ID")
                    end
                    
                    -- Если нашли ID — наносим урон
                    if zombieId then
                        ZombieDamage:FireServer(zombieId, 9999)
                        task.wait(DELAY_BETWEEN_KILLS)
                    end
                end
            end
        end
        task.wait(0.2) -- Небольшая пауза перед повторным сканированием папки
    end
end)

-- 4. Перехват спавна на лету
if _G.ZombieSpawnConnection then _G.ZombieSpawnConnection:Disconnect() end
_G.ZombieSpawnConnection = ZombieSpawn.OnClientEvent:Connect(function(zombieList)
    if not instakillEnabled or type(zombieList) ~= "table" then return end
    
    task.spawn(function()
        for _, subTable in ipairs(zombieList) do
            if not instakillEnabled then break end
            
            if type(subTable) == "table" then
                local zombieId = tonumber(subTable[1])
                if zombieId then
                    ZombieDamage:FireServer(zombieId, 99999)
                    task.wait(DELAY_BETWEEN_KILLS)
                end
            end
        end
    end)
end)

-- 5. Создание GUI
if PlayerGui:FindFirstChild("GodAutoFarmGui") then PlayerGui.GodAutoFarmGui:Destroy() end
local ScreenGui = Instance.new("ScreenGui", PlayerGui)
ScreenGui.Name = "GodAutoFarmGui"
ScreenGui.ResetOnSpawn = false

local MainFrame = Instance.new("Frame", ScreenGui)
MainFrame.Size = UDim2.new(0, 220, 0, 110)
MainFrame.Position = UDim2.new(0, 30, 0.5, -55)
MainFrame.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
Instance.new("UICorner", MainFrame).CornerRadius = UDim.new(0, 10)

local TitleLabel = Instance.new("TextLabel", MainFrame)
TitleLabel.Size = UDim2.new(1, 0, 0, 35)
TitleLabel.BackgroundTransparency = 1
TitleLabel.Text = "🚀 Smooth Instant Kill + Anti-AFK"
TitleLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
TitleLabel.TextSize = 14

local ToggleButton = Instance.new("TextButton", MainFrame)
ToggleButton.Size = UDim2.new(0, 180, 0, 50)
ToggleButton.Position = UDim2.new(0, 20, 0, 45)
ToggleButton.BackgroundColor3 = Color3.fromRGB(160, 40, 40)
ToggleButton.Text = "Destroying: off"
ToggleButton.TextColor3 = Color3.fromRGB(255, 255, 255)
ToggleButton.TextSize = 14
ToggleButton.Font = Enum.Font.SourceSansBold
Instance.new("UICorner", ToggleButton).CornerRadius = UDim.new(0, 8)

-- Скрипт перетаскивания GUI
local UserInputService = game:GetService("UserInputService")
local dragging, dragInput, dragStart, startPos

local function update(input)
    local delta = input.Position - dragStart
    MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then update(input) end
end)

-- Логика кнопки
ToggleButton.MouseButton1Click:Connect(function()
    instakillEnabled = not instakillEnabled
    if instakillEnabled then
        ToggleButton.BackgroundColor3 = Color3.fromRGB(40, 160, 40)
        ToggleButton.Text = "Destroying: on"
        task.wait(0.2)
        SkipVote:FireServer(true)
    else
        ToggleButton.BackgroundColor3 = Color3.fromRGB(160, 40, 40)
        ToggleButton.Text = "Destroying: off"
    end
end)

-- 6. НАДЕЖНЫЙ ANTI-AFK (Имитация кликов + Слот 1 раз в 5 минут)
if _G.AntiAFKTask then _G.AntiAFKTask = false end
_G.AntiAFKTask = true

-- Внутренний обход Idle-таймера через движок Roblox
if _G.AntiAFKConnection then _G.AntiAFKConnection:Disconnect() end
_G.AntiAFKConnection = LocalPlayer.Idled:Connect(function()
    VirtualUser:Button2Down(Vector2.new(0,0), Workspace.CurrentCamera.CFrame)
    task.wait(0.5)
    VirtualUser:Button2Up(Vector2.new(0,0), Workspace.CurrentCamera.CFrame)
end)

-- Цикл анимации "Нажатия на 1" (Раз в 5 минут)
task.spawn(function()
    while _G.AntiAFKTask do
        task.wait(600) -- Ждем 5 минут (300 секунд)
        
        -- Выполняем действие, только если автофарм включен и персонаж жив
        if instakillEnabled and LocalPlayer.Character then
            local backpack = LocalPlayer:FindFirstChild("Backpack")
            local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            
            if backpack and humanoid then
                local tools = backpack:GetChildren()
                if #tools > 0 then
                    -- Ищем самый первый инструмент в инвентаре (Аналог клавиши 1)
                    local firstTool = tools[1]
                    
                    if firstTool and firstTool:IsA("Tool") then
                        -- Эмуляция первого нажатия на "1" (Экипировать)
                        humanoid:EquipTool(firstTool)
                        
                        task.wait(2) -- Твоя задержка в 2 секунды
                        
                        -- Эмуляция второго нажатия на "1" (Убрать в рюкзак)
                        humanoid:UnequipTools()
                    end
                end
            end
        end
    end
end)
